### 2. Improve build.ps1
**Current state** (3 lines):
```powershell

```

**Improvements**:
```powershell
[CmdletBinding()]
param(
    [switch]$Clean,
    [switch]$NoBuild,  # Dev mode: just set up venv
    [string]$Config = ".
NewWorld Config Manager.spec"
)

# Error handling
$ErrorActionPreference = "Stop"

# Check for venv
if (!(Test-Path ".venv")) {
    Write-Error ".venv not found. Run setup script first."
    exit 1
}

# Activate
.\.venv\Scripts\activate.ps1

# Clean if requested
if ($Clean) {
    Write-Host "Cleaning build artifacts..." -ForegroundColor Yellow
    Remove-Item -Recurse -Force "dist", "build" -ErrorAction SilentlyContinue
    Write-Host "Clean complete." -ForegroundColor Green
}

# Build
Write-Host "Building New World Config Manager..." -ForegroundColor Cyan
$spec = Get-Content $Config -Raw

& .\.venv\Scripts\pyinstaller.exe $Config --clean

if ($LASTEXITCODE -ne 0) {
    Write-Error "Build failed with exit code $LASTEXITCODE"
    exit $LASTEXITCODE
}

# Verify
if (Test-Path "dist\NewWorld Config Manager.exe") {
    $version = Get-Content VERSION -Raw
    $exe = Get-Item "dist\NewWorld Config Manager.exe"
    Write-Host "Build successful!" -ForegroundColor Green
    Write-Host "  Version: $version"
    Write-Host "  Output: $($exe.FullName)"
    Write-Host "  Size: $([math]::Round($exe.Length / 1MB, 2)) MB"
} else {
    Write-Error "Build output not found!"
    exit 1
}
```
